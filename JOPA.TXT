# 📋 Git — Полная шпаргалка (29 заданий)

# 1. Создать локальный репозиторий в текущей папке
git init

# 2. Посмотреть статус текущего репозитория
git status

# 3. Что такое ветка и какая ветка является обычно основной
# Ветка — это изолированная линия разработки. Основная ветка обычно называется 'main' (ранее 'master').

# 4. Добавить файл в контекст, который будет коммититься (в индекс)
git add <имя_файла>

# 5. Создать коммит на основе текущего контекста с комментарием
git commit -m "Описание изменений"

# 6. Создать коммит, включающий изменения всех отслеживаемых (tracked) файлов
git commit -am "Описание изменений"

# 7. Посмотреть протокол (лог) коммитов
git log
git log --oneline  # компактный вид

# 8. Посмотреть информацию о текущих настройках
git config --list

# 9. Убрать файл из контекста (из индекса / staging area)
git restore --staged <имя_файла>
# (старая команда: git reset HEAD <имя_файла>)

# 10. Посмотреть изменения в файле по сравнению с последним коммитом
git diff <имя_файла>

# 11. Убрать изменения относительно последнего коммита из файла (полный откат)
git restore <имя_файла>
# (старая команда: git checkout -- <имя_файла>)

# 12. Добавить в контекст коммита все измененные и созданные файлы
git add .
# или
git add -A

# 13. Изменить глобальные/локальные настройки
git config --local <секция.ключ> <значение>   # только для этого репозитория
git config --global <секция.ключ> <значение>  # для всех репозиториев на ПК

# 14. Переписать имя пользователя
git config --global user.name "Твое Имя"
git config --global user.email "твой@email.com"

# 15. Просмотреть существующие ветки
git branch

# 16. Создать новую ветку
git branch <имя_новой_ветки>

# 17. Переключиться на другую ветку
git switch <имя_ветки>
# (старая команда: git checkout <имя_ветки>)

# 18. Создать новую ветку и сразу же переключиться на неё
git switch -c <имя_новой_ветки>
# (старая команда: git checkout -b <имя_новой_ветки>)

# 19. Удалить ветку
git branch -d <имя_ветки>   # безопасное удаление (только если примержена)
git branch -D <имя_ветки>   # принудительное удаление (даже если не примержена)

# 20. Примержить изменения из указанной ветки в текущую
git merge <имя_ветки-источника>

# 21. В каком случае могут появиться конфликты?
# Конфликт возникает, когда две разные ветки изменили одни и те же строки в одном и том же файле.

# 22. Как посмотреть, в каких файлах конфликты
git status
# (искать раздел "Unmerged paths" или "both modified")

# 23. Как устранить конфликты
# 1. Открыть файл, удалить маркеры конфликта (<<<<<<<, =======, >>>>>>>) и оставить нужный код.
# 2. git add <имя_файла>
# 3. git commit

# 24. Как переключиться на указанный коммит
git switch --detach <хэш_коммита>
# (старая команда: git checkout <хэш_коммита>)

# 25. Сделать ребазирование (rebase) текущей ветки
git rebase <имя_базовой_ветки>

# 26. Устранение конфликтов во время ребазирования
# 1. Исправить файл (удалить маркеры).
# 2. git add <имя_файла>
# 3. git rebase --continue

# 27. Отменить ребазирование во время конфликтов
git rebase --abort

# 28. Пропустить текущий конфликтный коммит и перейти к следующему
git rebase --skip

# 29. Отправить изменения из локального репозитория в удалённый
git push origin <имя_ветки>
# Для первой отправки новой ветки (связывает локальную и удаленную ветки):
git push -u origin <имя_ветки>


# ==============================================================================
# 🔧 ДОПОЛНИТЕЛЬНЫЕ ПОЛЕЗНЫЕ КОМАНДЫ
# ==============================================================================

# Показать подключенные удалённые репозитории
git remote -v

# Добавить удалённый репозиторий
git remote add origin <URL_репозитория>

# Изменить URL удалённого репозитория
git remote set-url origin <новый_URL>

# Скачать изменения с удалённого репозитория и сразу слить их с текущей веткой
git pull

# Скачать изменения с удалённого репозитория без слияния
git fetch

# Скачать репозиторий с нуля
git clone <URL_репозитория>
